cmake_minimum_required(VERSION 3.20)
project(rates_curve_nlp CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W3 /EHsc /permissive- /Zc:__cplusplus /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  add_compile_options(-Wall -Wextra)
endif()
add_library(rcmm INTERFACE)
target_include_directories(rcmm INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(rcmm_cli src/main.cpp)
target_link_libraries(rcmm_cli PRIVATE rcmm)
set_target_properties(rcmm_cli PROPERTIES OUTPUT_NAME rcmm)
add_executable(rcmm_tests tests/tests.cpp)
target_link_libraries(rcmm_tests PRIVATE rcmm)
enable_testing()
add_test(NAME rcmm_tests COMMAND rcmm_tests)
